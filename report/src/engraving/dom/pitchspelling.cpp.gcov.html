     412              : // The value of prefer sets the preferred mix of flats and sharps
     413              : // for pitches that are non-diatonic in the key specified, by
     414              : // positioning the window along the tpc sequence.
     415              : //
     416              : // Scale tones are the range shown in [ ].
     417              : // A value of 8 (Prefer::FLATS) specifies 5b 2b 6b 3b 7b [4 1 5 2 6 3 7]
     418              : // A value of 11 (Prefer::NEAREST) specifies 3b 7b [4 1 5 2 6 3 7] 4# 1# 5#
     419              : // A value of 13 (Prefer::SHARPS) specifies [4 1 5 2 6 3 7] 4# 1# 5# 2# 6#
     420              : //
     421              : // Examples for Prefer::NEAREST (n indicates explicit natural):
     422              : // C major will use Eb Bb [F C G D A E B] F# C# G#.
